- calculateKwp(areaM2): round(areaM2*0.17 to 1 decimal)
- calculateProduction(kwp): round(kwp*1150)
- calculateKlimabonus(kwp, hasBattery):
  if kwp <= 15 => 9300
  else => 9300 + (kwp-15)*620
  if hasBattery => +2250
  round to integer

  basic=2100/kWp, premium=2400/kWp

  profile multipliers low=0.35 normal=0.45 high=0.55
  savings = monthlyBill*12*multiplier (round int)
- calculatePayback(netCost, annualSavings): round((netCost/annualSavings) to 1 decimal)
